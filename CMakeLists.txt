cmake_minimum_required(VERSION 3.16)
project(BtechPort VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BTP_FRONTEND "Build the SDL2 native window" ON)
option(BTP_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(BTP_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()

add_library(btp_core port/core/game.c port/core/script.c port/core/talk.c port/core/npc.c port/core/shop.c port/core/bank.c port/core/economy.c port/core/school.c port/core/training.c port/core/animation.c port/data/resources.c port/data/anm.c port/data/sprites.c port/data/save.c)
target_include_directories(btp_core PUBLIC port)
target_sources(btp_core PRIVATE port/core/combat_rules.c port/core/combat.c)
add_library(btp_pixels port/frontend/pixels.c)
target_link_libraries(btp_pixels PUBLIC btp_core)
add_executable(btech-headless port/frontend/headless.c port/frontend/cli.c)
target_link_libraries(btech-headless PRIVATE btp_pixels)

if(BTP_FRONTEND)
  find_package(SDL2 QUIET CONFIG)
  if(TARGET SDL2::SDL2)
    set(BTP_SDL_TARGET SDL2::SDL2)
  else()
    find_path(BTP_SDL_INCLUDE SDL.h PATH_SUFFIXES SDL2
      HINTS "${PROJECT_SOURCE_DIR}/local-deps/sdl2/usr/include/SDL2")
    find_path(BTP_SDL_CONFIG SDL2/_real_SDL_config.h
      HINTS "${PROJECT_SOURCE_DIR}/local-deps/sdl2/usr/include/x86_64-linux-gnu")
    find_library(BTP_SDL_LIBRARY NAMES SDL2 SDL2-2.0)
    if(NOT BTP_SDL_LIBRARY AND UNIX)
      find_file(BTP_SDL_LIBRARY NAMES libSDL2-2.0.so.0
        PATHS /lib /usr/lib PATH_SUFFIXES x86_64-linux-gnu aarch64-linux-gnu)
    endif()
    if(NOT BTP_SDL_INCLUDE OR NOT BTP_SDL_LIBRARY)
      message(FATAL_ERROR "SDL2 development files not found. Install them or use -DBTP_FRONTEND=OFF.")
    endif()
    add_library(btp_sdl INTERFACE)
    target_include_directories(btp_sdl INTERFACE "${BTP_SDL_INCLUDE}")
    if(BTP_SDL_CONFIG)
      target_include_directories(btp_sdl INTERFACE "${BTP_SDL_CONFIG}")
    endif()
    target_link_libraries(btp_sdl INTERFACE "${BTP_SDL_LIBRARY}")
    set(BTP_SDL_TARGET btp_sdl)
  endif()
  add_executable(btech-port port/frontend/sdl.c port/frontend/cli.c)
  target_link_libraries(btech-port PRIVATE btp_pixels ${BTP_SDL_TARGET})
endif()
