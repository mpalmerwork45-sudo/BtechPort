if(BTP_FRONTEND)
  find_package(SDL2 QUIET CONFIG)
  if(TARGET SDL2::SDL2)
    set(BTP_SDL_TARGET SDL2::SDL2)
  else()
    find_path(BTP_SDL_INCLUDE SDL.h PATH_SUFFIXES SDL2
      HINTS "${PROJECT_SOURCE_DIR}/local-deps/sdl2/usr/include/SDL2")
    find_path(BTP_SDL_CONFIG SDL2/_real_SDL_config.h
      HINTS "${PROJECT_SOURCE_DIR}/local-deps/sdl2/usr/include/x86_64-linux-gnu")
    find_library(BTP_SDL_LIBRARY NAMES SDL2 SDL2-2.0)
    if(NOT BTP_SDL_LIBRARY AND UNIX)
      find_file(BTP_SDL_LIBRARY NAMES libSDL2-2.0.so.0
        PATHS /lib /usr/lib PATH_SUFFIXES x86_64-linux-gnu aarch64-linux-gnu)
    endif()
    if(NOT BTP_SDL_INCLUDE OR NOT BTP_SDL_LIBRARY)
      message(FATAL_ERROR "SDL2 development files not found. Install them or use -DBTP_FRONTEND=OFF.")
    endif()
    add_library(btp_sdl INTERFACE)
    target_include_directories(btp_sdl INTERFACE "${BTP_SDL_INCLUDE}")
    if(BTP_SDL_CONFIG)
      target_include_directories(btp_sdl INTERFACE "${BTP_SDL_CONFIG}")
    endif()
    target_link_libraries(btp_sdl INTERFACE "${BTP_SDL_LIBRARY}")
    set(BTP_SDL_TARGET btp_sdl)
  endif()
  add_executable(btech-port port/frontend/sdl.c port/frontend/cli.c)
  target_link_libraries(btech-port PRIVATE btp_pixels ${BTP_SDL_TARGET})
endif()
